  // src/services/authService.ts
  import { createClient } from '@supabase/supabase-js';
  import { appConfig } from '../config/app';

  class AuthService {
    private supabase = appConfig.supabase ?
      createClient(appConfig.supabase.url, appConfig.supabase.anonKey) : null;

    async login(email: string, password: string) {
      if (appConfig.mode === 'standalone') {
        return { success: true, user: null }; // 单用户模式直接通过
      }

      const { data, error } = await this.supabase!.auth.signInWithPassword({
        email,
        password,
      });

      return { success: !error, user: data.user, error };
    }

    async getCurrentUser() {
      if (appConfig.mode === 'standalone') {
        return null;
      }

      const { data } = await this.supabase!.auth.getUser();
      return data.user;
    }

    async logout() {
      if (appConfig.mode === 'standalone') return;

      await this.supabase!.auth.signOut();
    }
  }

  export const authService = new AuthService();

  // src/services/geminiService.ts 
  import { appConfig } from '../config/app';
  import { authService } from './authService';

  export class GeminiService {
    private async getApiEndpoint() {
      if (appConfig.mode === 'standalone') {
        // 直接调用 Gemini API
        return 'https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash-exp:generateContent';
      } else {
        // 通过后端代理
        return `${appConfig.backend!.url}/api/gemini/generate`;
      }
    }

    private async getHeaders() {
      const headers: Record<string, string> = {
        'Content-Type': 'application/json',
      };

      if (appConfig.mode === 'standalone') {
        headers['Authorization'] = `Bearer ${import.meta.env.VITE_GEMINI_API_KEY}`;
      } else {
        const user = await authService.getCurrentUser();
        if (user) {
          headers['Authorization'] = `Bearer ${user.access_token}`;
        }
      }

      return headers;
    }

        method: 'POST',
        headers,
        body: JSON.stringify(request),
      });

    }
  }

    // 验证用户 JWT token
    async verifyUser(token: string) {
      const { data, error } = await this.supabase.auth.getUser(token);
      return { user: data.user, error };
    }

    // 记录使用统计
    async recordUsage(userId: string, action: 'generate' | 'edit', tokenUsed: number) {
      const today = new Date().toISOString().split('T')[0];

      await this.supabase.rpc('increment_usage_stats', {
        p_user_id: userId,
        p_date: today,
        p_token_consumed: tokenUsed,
        p_action_type: action
      });
    }

    // 记录行为日志
    async logAction(userId: string, action: string, details: any, sessionId?: string) {
      await this.supabase
        .from('action_logs')
        .insert({
          user_id: userId,
          action,
          details,
          session_id: sessionId,
        });
    }

    // 保存聊天历史
    async saveChatHistory(userId: string, projectId: string, prompt: string, response: string, metadata: any) {
      await this.supabase
        .from('chat_history')
        .insert({
          user_id: userId,
          project_id: projectId,
          prompt,
          response,
          type: metadata.type,
          metadata,
        });
    }
  }

  export async function authMiddleware(req: Request, res: Response, next: NextFunction) {
    const token = req.headers.authorization?.replace('Bearer ', '');

    if (!token) {
      return res.status(401).json({ error: 'No token provided' });
    }

    const { user, error } = await supabaseService.verifyUser(token);

    if (error || !user) {
      return res.status(401).json({ error: 'Invalid token' });
    }

    req.user = user;
    next();
  }

  -- 这个脚本会在首次部署时运行，创建所有必要的表和策略

  -- 创建自定义函数用于统计更新
  CREATE OR REPLACE FUNCTION increment_usage_stats(
      p_user_id UUID,
      p_date DATE,
      p_token_consumed INTEGER DEFAULT 0,
      p_action_type TEXT DEFAULT 'generate'
  ) RETURNS VOID AS $$
  BEGIN
      INSERT INTO usage_stats (user_id, date, token_consumed, request_count, generation_count, edit_count)
      VALUES (p_user_id, p_date, p_token_consumed, 1,
              CASE WHEN p_action_type = 'generate' THEN 1 ELSE 0 END,
              CASE WHEN p_action_type = 'edit' THEN 1 ELSE 0 END)
      ON CONFLICT (user_id, date)
      DO UPDATE SET
          token_consumed = usage_stats.token_consumed + p_token_consumed,
          request_count = usage_stats.request_count + 1,
          generation_count = usage_stats.generation_count + (CASE WHEN p_action_type = 'generate' THEN 1 ELSE 0
  END),
          edit_count = usage_stats.edit_count + (CASE WHEN p_action_type = 'edit' THEN 1 ELSE 0 END);
  END;
  $$ LANGUAGE plpgsql SECURITY DEFINER;

    const token = req.headers.authorization?.replace('Bearer ', '');
    if (!token) {
      return res.status(401).json({ error: 'No token provided' });
    }

    try {
      const { data: { user }, error } = await supabase.auth.getUser(token);
      if (error || !user) {
        return res.status(401).json({ error: 'Invalid token' });
      }
      req.user = user;
      next();
    } catch (error) {
      res.status(500).json({ error: 'Auth verification failed' });
    }
  }

  // Gemini 生成代理
  router.post('/generate', authMiddleware, async (req, res) => {
    try {
      const { prompt, parameters } = req.body;
      const userId = req.user.id;

      // 调用 Gemini API (这里需要集成实际的 Gemini 调用逻辑)
      const result = await callGeminiAPI({ prompt, ...parameters });

      // 记录使用统计
      if (supabase && process.env.APP_MODE === 'multi-user') {
        await recordUsage(userId, 'generate', estimateTokens(prompt));
        await logAction(userId, 'generate', { prompt, parameters });
      }

      res.json({ success: true, data: result });
    } catch (error) {
      res.status(500).json({ error: 'Generation failed' });
    }
  });
